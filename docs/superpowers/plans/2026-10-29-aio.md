```ts
import type { Article, Block } from './types'

/**
 * An article as Markdown, for the `.md` twin an AI agent reads instead of the
 * HTML (`/vi/blog/<id>.md`, `/vi/learn/<id>.md`). Every block type is
 * handled by the switch below; a new one fails to compile here first.
 */

/** A line that would start a Markdown construct is escaped. */
const line = (s: string) => s.replace(/^(\s*)([#>\-+*]|\d+\.)/, '$1\\$2')
const cell = (s: string) => s.replace(/\|/g, '\\|').replace(/\n/g, ' ')

function blockMarkdown(b: Block): string {
  switch (b.t) {
    case 'tldr':
      return b.items.map((i) => `> ${i}`).join('\n')
    case 'h':
      return `## ${b.text}`
    case 'p':
      return line(b.text)
    case 'menu':
      return `- Cũ: ${b.old}\n- Mới: ${b.new}`
    case 'table':
      return [
        `| ${b.head.map(cell).join(' | ')} |`,
        `| ${b.head.map(() => '---').join(' | ')} |`,
        ...b.rows.map((r) => `| ${r.map(cell).join(' | ')} |`),
        ...(b.caption ? ['', `*${b.caption}*`] : []),
      ].join('\n')
    case 'callout':
      return `> **${b.label}** ${b.text}`
    case 'compare':
      return `*So sánh ảnh: ${b.beforeLabel} / ${b.afterLabel} — ${b.caption}*`
    case 'checklist':
      return `**${b.label}**\n\n${b.items.map((i) => `- [ ] ${i}`).join('\n')}`
    case 'figure':
      return `*Hình: ${b.caption}*`
    case 'embed':
      return `[${b.caption}](${b.provider === 'youtube' ? `https://www.youtube.com/watch?v=${b.id}` : `https://vimeo.com/${b.id}`})`
    default: {
      const never: never = b
      return never
    }
  }
}

/** The words a reader sees in one block — for `wordCount`. */
export function plainText(b: Block): string {
  switch (b.t) {
    case 'tldr':
      return b.items.join(' ')
    case 'h':
    case 'p':
      return b.text
    case 'menu':
      return `${b.old} ${b.new}`
    case 'table':
      return [...b.head, ...b.rows.flat(), b.caption].join(' ')
    case 'callout':
      return `${b.label} ${b.text}`
    case 'compare':
      return `${b.beforeLabel} ${b.afterLabel} ${b.caption}`
    case 'checklist':
      return [b.label, ...b.items].join(' ')
    case 'figure':
    case 'embed':
      return b.caption
    default: {
      const never: never = b
      return never
    }
  }
}

export function articleMarkdown(article: Article, opts: { canonicalUrl: string }): string {
  const head = [`# ${article.title}`, line(article.dek), `Nguồn gốc: ${opts.canonicalUrl}`]
  const body = article.blocks.map(blockMarkdown)
  const sources = article.meta.sources.length
    ? [
        '## Nguồn',
        article.meta.sources
          .map((s) => `- [${s.title}](${s.url})${[s.publisher, s.checkedAt].filter(Boolean).length ? ` — ${[s.publisher, s.checkedAt].filter(Boolean).join(', ')}` : ''}`)
          .join('\n'),
      ]
    : []
  return [...head, ...body, ...sources].filter((s) => s !== '').join('\n\n') + '\n'
}
```

```ts
import { conceptTitle } from '@/lib/content/concepts'
import { conceptAnchor } from '@/lib/learn/glossary'
import { ARTICLE_LANG } from './articles'
import { plainText } from './markdown'
import type { Article, Block } from './types'

/** Question headings each answered by the paragraph right after them — the page's own words. */
export function faqPairs(blocks: readonly Block[]): { question: string; answer: string }[] {
  const out: { question: string; answer: string }[] = []
  blocks.forEach((b, i) => {
    const next = blocks[i + 1]
    if (b.t === 'h' && b.text.trim().endsWith('?') && next?.t === 'p') out.push({ question: b.text.trim(), answer: next.text })
  })
  return out
}

/**
 * The JSON-LD nodes for one article or knowledge page. See the header of
 * `structured-data.tsx` for what is deliberately not declared (no
 * `datePublished`; the reviewed date stays on the page).
 */
export function articleJsonLd(article: Article, opts: { site: string; path: string; learnTitle: string }): Record<string, unknown>[] {
  const { site } = opts
  const url = `${site}${opts.path}`
  const vi = `${site}/${ARTICLE_LANG}`
  const knowledge = article.kind === 'knowledge'
  const parent = knowledge ? { name: opts.learnTitle, item: `${vi}/learn` } : { name: 'Alpha Tech Blogs', item: `${vi}/blog` }
  const concepts = article.meta.concepts
  const words = article.blocks.map(plainText).join(' ').split(/\s+/).filter(Boolean).length
  const faq = faqPairs(article.blocks)

  const nodes: Record<string, unknown>[] = [
    {
      '@context': 'https://schema.org',
      '@type': knowledge ? 'TechArticle' : 'BlogPosting',
      headline: article.title,
      ...(article.dek ? { description: article.dek } : {}),
      url,
      mainEntityOfPage: url,
      inLanguage: ARTICLE_LANG,
      ...(article.updatedAt ? { dateModified: article.updatedAt } : {}),
      ...(article.meta.authorName ? { author: { '@type': 'Person', name: article.meta.authorName } } : {}),
      publisher: { '@type': 'Organization', name: 'Alpha ColorLab', url: site, logo: `${site}/icon` },
      ...(concepts.length
        ? {
            keywords: concepts.map(conceptTitle).join(', '),
            about: concepts.map((k) => ({
              '@type': 'DefinedTerm',
              name: conceptTitle(k),
              url: `${vi}/learn/glossary#${conceptAnchor(k)}`,
              inDefinedTermSet: `${vi}/learn/glossary`,
            })),
          }
        : {}),
      wordCount: words,
      ...(article.meta.sources.length
        ? {
            citation: article.meta.sources.map((s) => ({
              '@type': 'CreativeWork',
              name: s.title,
              url: s.url,
              ...(s.publisher ? { publisher: { '@type': 'Organization', name: s.publisher } } : {}),
            })),
          }
        : {}),
      isPartOf: { '@type': 'WebSite', name: 'Alpha ColorLab', url: site },
    },
    {
      '@context': 'https://schema.org',
      '@type': 'BreadcrumbList',
      itemListElement: [
        { '@type': 'ListItem', position: 1, name: parent.name, item: parent.item },
        { '@type': 'ListItem', position: 2, name: article.title, item: url },
      ],
    },
  ]
  if (faq.length) {
    nodes.push({
      '@context': 'https://schema.org',
      '@type': 'FAQPage',
      mainEntity: faq.map((q) => ({ '@type': 'Question', name: q.question, acceptedAnswer: { '@type': 'Answer', text: q.answer } })),
    })
  }
  return nodes
}
```

```ts
import { getPublishedArticle, getPublishedKnowledgePage } from '@/lib/lab/data'
import { ARTICLE_LANG } from '@/lib/lab/articles'
import { articleMarkdown } from '@/lib/lab/markdown'

/**
 * The Markdown twin of a published page, for AI agents. Reached through the
 * rewrites in `next.config.ts` from `/vi/blog/<id>.md` and
 * `/vi/learn/<id>.md` — the canonical URL plus `.md`. Drafts and unknown ids
 * are 404s: the published readers return nothing for them. A failed online
 * read throws, as the pages do.
 */
export const revalidate = 3600

const SITE = process.env.NEXT_PUBLIC_SITE_URL ?? 'http://localhost:3000'
const READERS = { article: getPublishedArticle, knowledge: getPublishedKnowledgePage } as const
const PREFIX = { article: 'blog', knowledge: 'learn' } as const

export async function GET(_req: Request, ctx: { params: Promise<{ kind: string; id: string }> }) {
  const { kind, id } = await ctx.params
  if (kind !== 'article' && kind !== 'knowledge') return new Response('Not found', { status: 404 })
  const page = await READERS[kind](id)
  if (!page) return new Response('Not found', { status: 404 })

  return new Response(articleMarkdown(page, { canonicalUrl }), {
    headers: { 'content-type': 'text/markdown; charset=utf-8' },
  })
}
```

```ts
import { ARTICLE_LANG } from '@/lib/lab/articles'
import type { Article } from '@/lib/lab/types'

/** The whole site as an llms.txt (llmstxt.org): a title, a summary, then one line per page. */
export type LlmsInput = {
  site: string
  knowledge: readonly Article[]
  articles: readonly Article[]
  recipes: readonly { name: string; slug: string; description: string }[]
  cameras: readonly { id: string; name: string }[]
  audio: readonly { id: string; name: string }[]
}

export const firstSentence = (s: string) => (/^.*?[.!?](?=\s|$)/.exec(s.trim())?.[0] ?? s.trim())

const item = (name: string, url: string, note?: string) => `- [${name}](${url})${note ? `: ${note}` : ''}`

export function llmsTxt(i: LlmsInput): string {
  const vi = `${i.site}/${ARTICLE_LANG}`
  const section = (title: string, lines: string[]) => [`## ${title}`, '', ...lines].join('\n')
  return [
    '# Alpha ColorLab',
    '> Colour recipes for Sony Alpha cameras — White Balance with a Picture Profile or a Creative Look — plus a Sony camera, lens and audio catalogue and Vietnamese articles on camera settings. Recipe names and camera terms are as printed on the camera.',
    section('Learn', i.knowledge.map((p) => item(p.title, `${vi}/learn/${p.id}.md`, p.dek))),
    section('Blog', i.articles.map((p) => item(p.title, `${vi}/blog/${p.id}.md`, p.dek))),
    section('Recipes', i.recipes.map((r) => item(r.name, `${i.site}/recipe/${r.slug}`, firstSentence(r.description)))),
    section('Cameras & lenses', i.cameras.map((c) => item(c.name, `${i.site}/cameras/${c.id}`))),
    section('Audio', i.audio.map((a) => item(a.name, `${i.site}/audio/${a.id}`))),
    section('Hubs', [
      item('Recipe catalogue', `${i.site}/colorlab`),
      item('Cameras & lenses', `${i.site}/cameras`),
      item('Audio', `${i.site}/audio`),
      item('Learn', `${vi}/learn`),
      item('Glossary', `${vi}/learn/glossary`),
      item('Blog', `${vi}/blog`),
      item('Search', `${i.site}/search`),
    ]),
  ].join('\n\n') + '\n'
}
```
